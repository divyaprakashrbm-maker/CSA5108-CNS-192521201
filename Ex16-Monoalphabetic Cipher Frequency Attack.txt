#include<stdio.h>
int main(){
    char c[200]; int i;
    printf("Enter ciphertext: ");
    scanf("%199s",c);
    printf("Ciphertext: %s\n",c);
    printf("Use letter frequency to estimate plaintext.\n");
    for(i=0;c[i];i++)
        if(c[i]>='A'&&c[i]<='Z')
            printf("%c:%d ",c[i],c[i]-'A');
    return 0;
}
